    <!-- JAVASCRIPT CONTROL -->

    <script>

        const modeToggle = document.getElementById('modeToggle');

        const introSection = document.getElementById('introduction');

        const dictSection = document.getElementById('dictionary');

        const wordDetail = document.getElementById('wordDetail');


        // Dữ liệu mẫu cho Từ điển (Bạn có thể sửa nội dung và tên file ảnh tại đây)

        const dictionaryData = {

            'A': { ko: '추억 / 기억', en: 'Memory / Remembrance', img: 'images/a-art.png' },

            'B': { ko: '상자', en: 'Box', img: 'images/b-art.png' },

            'C': { ko: '노래', en: 'Song', img: 'images/c-art.png' },

            // Thêm các ký tự khác tương tự vào đây...

        };


        // Xử lý sự kiện Gạt nút (Chuyển Tab & Đổi màu nền)

        modeToggle.addEventListener('change', function() {

            if (this.checked) {

                // Sang phải: Dictionary (Nền đen chữ trắng)

                document.documentElement.setAttribute('data-theme', 'dark');

                introSection.classList.remove('active');

                dictSection.classList.add('active');

            } else {

                // Sang trái: Introduction (Nền trắng chữ đen)

                document.documentElement.removeAttribute('data-theme');

                dictSection.classList.remove('active');

                introSection.classList.add('active');

            }

        });


        // Xử lý sự kiện click vào từng ký tự chữ cái

        function showWord(letter) {

            // Active hiệu ứng nút được chọn

            const buttons = document.querySelectorAll('.letter-btn');

            buttons.forEach(btn => {

                if(btn.textContent === letter) btn.classList.add('active');

                else btn.classList.remove('active');

            });


            // Lấy dữ liệu chữ cái tương ứng

            const data = dictionaryData[letter];


            if (data) {

                document.getElementById('displayLetter').textContent = letter;

                document.getElementById('displayKo').textContent = data.ko;

                document.getElementById('displayEn').textContent = data.en;

                document.getElementById('displayImg').src = data.img;

                document.getElementById('displayImg').alt = `Artwork for ${letter}`;

                wordDetail.style.display = 'block';

            } else {

                // Trường hợp ký tự chưa được nhập dữ liệu cụ thể

                document.getElementById('displayLetter').textContent = letter;

                document.getElementById('displayKo').textContent = '준비 중인 단어입니다.';

                document.getElementById('displayEn').textContent = 'Content is coming soon.';

                document.getElementById('displayImg').src = '';

                document.getElementById('displayImg').alt = 'No image available';

                wordDetail.style.display = 'block';

            }

        }

    </script>
